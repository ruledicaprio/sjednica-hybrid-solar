import os

def merge_files_for_radiance(source_folder, output_filename="spojeni_dokumenti.txt"):
    # Određivanje putanje roditeljskog foldera
    parent_folder = os.path.dirname(source_folder)
    output_path = os.path.join(parent_folder, output_filename)
    
    # Podržani formati (dodaj ili oduzmi po potrebi)
    extensions = ('.txt', '.csv', '.log', '.md', '.py', '.json', '.html', '.css', '.js')

    print(f"Skeniram folder: {source_folder}")
    print(f"Rezultat će biti spremljen u: {output_path}\n")

    try:
        with open(output_path, 'w', encoding='utf-8') as outfile:
            for filename in os.listdir(source_folder):
                if filename.endswith(extensions):
                    file_path = os.path.join(source_folder, filename)
                    
                    try:
                        with open(file_path, 'r', encoding='utf-8') as infile:
                            outfile.write(f"\n{'#'*60}\n")
                            outfile.write(f" DATOTEKA: {filename}\n")
                            outfile.write(f"{'#'*60}\n\n")
                            
                            outfile.write(infile.read())
                            outfile.write("\n")
                        
                        print(f"Uspješno dodano: {filename}")
                    except Exception as e:
                        print(f"Preskočeno (greška): {filename} - {e}")
        
        print(f"\nGotovo! Datoteka se nalazi na: {output_path}")

    except Exception as e:
        print(f"Kritična greška pri kreiranju izlazne datoteke: {e}")

# Direktna putanja do tvog foldera
tvoj_folder = r"E:\Radiance_build\rural-star-sjednica"

merge_files_for_radiance(tvoj_folder)
